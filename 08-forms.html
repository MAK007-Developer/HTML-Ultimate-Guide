    08 · Forms

    Forms

    The longest section of this guide — forms are where accessibility, usability, security and browser quirks collide.

    Structure: fieldset, legend, label, name, button

    <form action="/subscribe" method="post">
  <fieldset>
    <legend>Newsletter sign-up</legend>

    <p>
      <label for="email">Email address</label>
      <input id="email" name="email" type="email"
             autocomplete="email" required
             aria-describedby="email-hint">
      <span id="email-hint">We'll never share it.</span>
    </p>

    <p>
      <button type="submit">Subscribe</button>
    </p>
  </fieldset>
</form>
    • fieldset + legend group related controls and give the group an accessible name.
    • label + for must match the input's id. Clicking the label focuses the control.
    • name is what actually gets submitted. A control without name sends nothing.
    • type="submit" on every button — the default is submit, and an untyped button in a form will submit it when clicked.

    form attributes

    Form attributes
    AttributeValuesWhat it does
    actionURLWhere the form submits. Omit to submit to the current URL.
    methodget / postget puts data in the URL (search forms); post puts it in the body (state changes).
    enctypeurlencoded, multipart/form-data, text/plainRequired for file uploads: multipart/form-data.
    novalidatebooleanDisables the browser's built-in validation UI.
    autocompleteon / offHints for the whole form; per-input tokens are far more useful.
    target_self, _blank, nameWhere to display the response. Pair with rel="noopener".
    relnoopener, noreferrerSecurity for target="_blank".
    accept-charsetencoding nameRarely needed; UTF-8 is the norm.
    namestringNames the form for access via document.forms.

    The type="number" caveat

    <!-- ❌ type="number" is a bad fit for things that *look* numeric
     but aren't quantities: phone numbers, postal codes, credit cards,
     OTPs. It strips leading zeros, allows "e", "E", "+" and "-", and
     silently converts on scroll. -->
<input type="number" name="zip">

<!-- ✅ Use a text input with a numeric keyboard hint -->
<input type="text" name="zip"
       inputmode="numeric"
       pattern="[0-9]{5}"
       autocomplete="postal-code"
       maxlength="5">

<!-- inputmode values: numeric, decimal, tel, email, url, search, text -->

    Validation attributes

    <input type="text"
       name="username"
       required
       minlength="3"
       maxlength="20"
       pattern="[a-zA-Z0-9_]{3,20}">

<input type="number" name="qty" min="1" max="99" step="1">

<input type="file" name="avatar" accept="image/png, image/jpeg">

<input type="email" name="email" multiple>

<!-- pattern is matched against the *entire* value, anchored implicitly.
     Escape regex special chars carefully inside the attribute. -->

    Labels, hints and errors

    <div class="field">
  <label for="pw">Password</label>
  <input id="pw" name="password" type="password"
         autocomplete="new-password"
         required
         minlength="12"
         aria-describedby="pw-hint pw-error"
         aria-invalid="true">
  <p id="pw-hint">At least 12 characters.</p>
  <p id="pw-error" role="alert">
    Password must contain at least one number.
  </p>
</div>

<!-- aria-describedby points at BOTH the hint and the error.
     role="alert" announces the error as soon as it appears.
     aria-invalid="true" tells AT the field's current state. -->

    Checkboxes and radios

    <!-- Checkbox: independent booleans -->
<fieldset>
  <legend>Notifications</legend>

  <label>
    <input type="checkbox" name="notify" value="email" checked>
    Email
  </label>

  <label>
    <input type="checkbox" name="notify" value="sms">
    SMS
  </label>
</fieldset>

<!-- Radio: one-of-many, grouped by shared name -->
<fieldset>
  <legend>Plan</legend>

  <label>
    <input type="radio" name="plan" value="free" checked>
    Free
  </label>

  <label>
    <input type="radio" name="plan" value="pro">
    Pro
  </label>
</fieldset>

<!-- Wrapping the input inside the label is the most robust
     association method — no for/id required. -->

    select, optgroup, datalist, textarea

    <!-- select -->
<label for="country">Country</label>
<select id="country" name="country" autocomplete="country">
  <option value="">Choose…</option>
  <optgroup label="Europe">
    <option value="fr">France</option>
    <option value="de">Germany</option>
  </optgroup>
  <optgroup label="North America">
    <option value="us">United States</option>
    <option value="ca">Canada</option>
  </optgroup>
</select>

<!-- multiple -->
<select name="tags" multiple size="4">
  <option value="html" selected>HTML</option>
  <option value="css">CSS</option>
  <option value="js">JavaScript</option>
</select>

<!-- datalist: free text with suggestions -->
<label for="browser">Browser</label>
<input id="browser" name="browser" list="browsers" autocomplete="off">
<datalist id="browsers">
  <option value="Chrome">
  <option value="Firefox">
  <option value="Safari">
  <option value="Edge">
</datalist>

<!-- textarea -->
<label for="bio">Bio</label>
<textarea id="bio" name="bio" rows="6" maxlength="500"
          placeholder="Tell us about yourself…"></textarea>
<p id="bio-hint"><small>Max 500 characters.</small></p>

    Buttons and form overrides

    <form action="/save" method="post">
  <button type="submit">Save</button>

  <!-- Override the form's action, method and validation per button -->
  <button type="submit"
          formaction="/save-draft"
          formmethod="post"
          formnovalidate>
    Save draft
  </button>

  <!-- A non-submitting button MUST have type="button" -->
  <button type="button" onclick="preview()">Preview</button>

  <!-- Reset buttons are usually a bad idea -->
  <button type="reset">Reset</button>
</form>

    Hidden fields and CSRF

    <form action="/transfer" method="post">
  <!-- CSRF token — rendered server-side, unique per session -->
  <input type="hidden" name="csrf_token" value="a8f3…">

  <!-- Other hidden state (IDs, return URLs, version numbers) -->
  <input type="hidden" name="return_to" value="/dashboard">
  <input type="hidden" name="version" value="3">

  <button type="submit">Transfer</button>
</form>

<!-- Never put secrets in hidden fields: they are visible in the DOM.
     Verify the CSRF token server-side on every state-changing request. -->

    output, meter, progress

    <!-- output: a calculated result -->
<form oninput="total.value = Number(a.value) + Number(b.value)">
  <input id="a" name="a" type="number" value="0">
  +
  <input id="b" name="b" type="number" value="0">
  =
  <output name="total" for="a b">0</output>
</form>

<!-- meter: a scalar measurement within a known range -->
<label for="disk">Disk usage</label>
<meter id="disk" min="0" max="100" low="30" high="80" optimum="20" value="72">
  72%
</meter>

<!-- progress: completion of a task -->
<label for="upload">Uploading…</label>
<progress id="upload" max="100" value="45">45%</progress>

<!-- Indeterminate progress -->
<progress>Loading…</progress>

    The <search> landmark

    <!-- <search> is a landmark element for site-wide search forms -->
<search>
  <form action="/search" method="get" role="search">
    <label for="q">Search the site</label>
    <input id="q" name="q" type="search" placeholder="Search…">
    <button type="submit">Search</button>
  </form>
</search>
