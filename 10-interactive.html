    Interactive & Disclosure Elements

    Built-in interactive widgets that do the accessibility work for you — and the one thing you should use instead of a custom modal.

    details / summary

    <!-- Basic disclosure -->
<details>
  <summary>What is HTML?</summary>
  <p>HyperText Markup Language — the structure of the web.</p>
</details>

<!-- Open by default -->
<details open>
  <summary>Shipping</summary>
  <p>Ships in 2–3 business days.</p>
</details>

<!-- Exclusive accordion: only one open at a time (shared name) -->
<details name="faq">
  <summary>Returns</summary>
  <p>30-day return window.</p>
</details>
<details name="faq">
  <summary>Warranty</summary>
  <p>2-year manufacturer warranty.</p>
</details>

<!-- The summary is the keyboard-focusable control. Native toggle
     events fire on the details element when its state changes. -->

    dialog

    <!-- Native modal dialog: showModal() gives you focus trapping,
     ESC-to-close, and a ::backdrop pseudo-element for free. -->
<dialog id="confirm">
  <form method="dialog">
    <p>Delete this file?</p>
    <button value="cancel">Cancel</button>
    <button value="ok">Delete</button>
  </form>
</dialog>

<button id="open">Delete…</button>

<script type="module">
  const dlg = document.getElementById("confirm");
  const opener = document.getElementById("open");

  opener.addEventListener("click", () => {
    dlg.showModal();          // NOT show() — show() is non-modal
  });

  dlg.addEventListener("close", () => {
    if (dlg.returnValue === "ok") {
      console.log("deleted");
    }
    // Return focus to the element that opened the dialog.
    opener.focus();
  });
</script>

<style>
  dialog::backdrop { background: rgba(1,4,9,.55); }
</style>

    Popover API

    <!-- A popover is an element that can be shown/hidden from a
     trigger button. The browser handles focus, ESC, and outside-click. -->
<button popovertarget="menu">Open menu</button>

<div id="menu" popover>
  <ul>
    <li><a href="/profile">Profile</a></li>
    <li><a href="/settings">Settings</a></li>
    <li><a href="/logout">Log out</a></li>
  </ul>
</div>

<!-- popover="manual" disables light-dismiss (click outside doesn't close) -->
<div id="notice" popover="manual">Dismiss me with a button.</div>

<!-- Programmatic control -->
<script>
  const menu = document.getElementById("menu");
  menu.showPopover();   // or hidePopover() / togglePopover()
</script>

    template and slot

    <!-- <template> is inert: it is parsed but not rendered.
     Its content is available via .content for cloning. -->
<template id="row-template">
  <tr>
    <td class="name"></td>
    <td class="email"></td>
  </tr>
</template>

<script>
  const tpl = document.getElementById("row-template");
  const row = tpl.content.cloneNode(true);
  row.querySelector(".name").textContent = "Ada Lovelace";
  row.querySelector(".email").textContent = "ada@example.com";
  document.querySelector("tbody").appendChild(row);
</script>

     user-supplied children into the component's tree. -->
