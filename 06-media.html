    06 · Media

    Media

    Video, audio, iframes, canvas and inline SVG — including the security settings that actually matter.

    video

    <video controls
       width="640" height="360"
       poster="/img/poster.jpg"
       preload="metadata"
       playsinline
       crossorigin="anonymous">
  <source src="/media/clip.webm" type="video/webm">
  <source src="/media/clip.mp4"  type="video/mp4">

  <!-- Captions are not optional for spoken content -->
  <track kind="captions" src="/media/clip.en.vtt" srclang="en" label="English" default>
  <track kind="subtitles" src="/media/clip.fr.vtt" srclang="fr" label="Français">
  <track kind="chapters"  src="/media/chapters.vtt" srclang="en" label="Chapters">

  <p>Your browser can't play this video.
     <a href="/media/clip.mp4">Download the MP4</a>.</p>
</video>

<!-- Autoplay only works if the video is muted (and is hostile to users) -->
<video autoplay muted loop playsinline src="/media/bg.mp4"></video>

<!-- playsinline stops iOS from hijacking into fullscreen -->
<!-- preload="metadata" fetches duration/dimensions but not the whole file -->

    audio

    <audio controls preload="metadata">
  <source src="/media/episode-12.mp3" type="audio/mpeg">
  <source src="/media/episode-12.ogg" type="audio/ogg">
  <p>Your browser can't play audio.
     <a href="/media/episode-12.mp3">Download the episode</a>.</p>
</audio>

<!-- Provide a transcript link alongside the player for accessibility. -->

    iframe

    <iframe src="https://player.example.com/embed/abc123"
        title="Product tour video"
        width="640" height="360"
        loading="lazy"
        referrerpolicy="strict-origin-when-cross-origin"
        sandbox="allow-scripts allow-popups allow-presentation"
        allow="fullscreen; picture-in-picture"
        allowfullscreen></iframe>

<!-- title is REQUIRED: it is the frame's accessible name and shows
     when the frame fails to load. -->
    ✕

    Never combine allow-scripts with allow-same-origin on untrusted content. Together they let the framed document remove its own sandbox attribute and escape the sandbox entirely.

    canvas and inline SVG

    <!-- Canvas needs an accessible name + a text fallback -->
<canvas id="sales-chart" width="600" height="400"
        role="img"
        aria-label="Bar chart: sales rose from 12k in January to 31k in June">
  <p>Sales rose from 12k in January to 31k in June.
     <a href="/data/sales.csv">Download the raw data</a>.</p>
</canvas>

<!-- Inline SVG: give it a role and a title when meaningful -->
<svg role="img" aria-labelledby="warn-title" viewBox="0 0 24 24" width="24" height="24">
  <title id="warn-title">Warning</title>
  <path d="M12 2 1 21h22L12 2zm0 6v7m0 3v.5" fill="none" stroke="currentColor"/>
</svg>

<!-- Decorative SVG: hide it completely -->
<svg aria-hidden="true" focusable="false" viewBox="0 0 24 24" width="16" height="16">
  <path d="M4 4l16 16M20 4L4 20"/>
</svg>
