    Custom Elements

    <!-- Hyphen in the name is mandatory — it prevents collisions
     with built-in elements. Must be loaded as type="module". -->
<user-card user-id="42"></user-card>

<script type="module">
  class UserCard extends HTMLElement {
    // Which attributes trigger attributeChangedCallback
    static get observedAttributes() { return ["user-id"]; }

    connectedCallback() {
      this.render();
    }

    attributeChangedCallback(name, oldValue, newValue) {
      if (oldValue !== newValue) this.render();
    }

    render() {
      const id = this.getAttribute("user-id") ?? "—";
      this.textContent = `User #${id}`;
    }
  }
  customElements.define("user-card", UserCard);
</script>

    Shadow DOM

    <script type="module">
  class FancyButton extends HTMLElement {
    constructor() {
      super();
      const root = this.attachShadow({ mode: "open" });

      root.innerHTML = `
        <style>
          :host { display: inline-block; }
          :host([disabled]) { opacity: .5; pointer-events: none; }
          button { font: inherit; padding: .5rem 1rem;
                   border-radius: 6px; border: 1px solid #ccc;
                   background: #fff; }
          ::slotted(span) { font-weight: 700; }
          button::part(label) { letter-spacing: .02em; }
        </style>
        <button part="button"><slot name="label">Click</slot></button>
      `;
    }
  }
  customElements.define("fancy-button", FancyButton);
</script>

<fancy-button>
  <span slot="label">Save</span>
</fancy-button>
    • :host styles the custom element itself from inside the shadow root.
    • :host([attr]) matches when the host has a given attribute.
    • ::slotted(selector) styles light-DOM children projected into slots.
    • ::part(name) lets a shadow part be styled from outside the component.

    Declarative Shadow DOM

    <!-- Server-rendered shadow DOM, no JS required to attach.
     template shadowrootmode="open" creates the shadow root during
     HTML parsing; children of the template become the shadow content. -->

    <style>

    </style>

    Microdata vs JSON-LD

    Structured data formats
    FormatWhere it livesVerdict
    MicrodataInline itemscope / itemprop attributes in your markupCouples markup and data; harder to maintain.
    MicroformatsClass names (h-card, h-entry)Niche; not widely consumed.
    RDFaInline attributesOverkill for most sites.
    JSON-LDA <script type="application/ld+json"> blockPreferred. Decoupled from markup, easy to generate server-side, easy to validate.

    Security: SRI, referrer, CSP

    <!-- Subresource Integrity: pin a CDN file to an exact hash.
     If the file changes, the browser refuses to execute it. -->
<script src="https://cdn.example.com/lib.js"
        integrity="sha384-BASE64HASHHERE"
        crossorigin="anonymous"></script>

<!-- Cross-origin stylesheet SRI -->
<link rel="stylesheet"
      href="https://cdn.example.com/style.css"
      integrity="sha384-BASE64HASHHERE"
      crossorigin="anonymous">

<!-- Referrer policy: send full URL to same-origin, only origin to
     cross-origin, nothing on downgrade. -->
<meta name="referrer" content="strict-origin-when-cross-origin">

<!-- CSP via meta tag — a nonce lets you whitelist one inline script
     without enabling all of them. -->
<meta http-equiv="Content-Security-Policy"
      content="default-src 'self';
               script-src 'self' 'nonce-rAnd0m123';
               style-src  'self' 'unsafe-inline';
               img-src    'self' https://cdn.example.com data:;">
<script nonce="rAnd0m123">
  // this inline script is allowed by the nonce above
</script>

<!-- ⚠️  NEVER use meta refresh to redirect to an untrusted URL —
     it is a common open-redirect and phishing vector. Use a server
     301/302 instead. -->
